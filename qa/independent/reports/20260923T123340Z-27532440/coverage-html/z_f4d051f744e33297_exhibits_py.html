1"""Where a plain centrality ranking and an evidence-based ranking disagree. 

2 

3PageRank answers "which node does flow concentrate on". It does not distinguish a node 

4collecting from many payers from a node paying many suppliers. On this data the two 

5rankings share nothing at the top, so the disagreement is worth showing rather than 

6hiding behind a single number. 

7""" 

8from __future__ import annotations 

9 

10import pandas as pd 

11 

12 

13def pagerank_vs_evidence(feats: pd.DataFrame, top_n: int = 15) -> pd.DataFrame: 

14 df = feats.copy() 

15 df["rank_pagerank"] = df.pagerank.rank(ascending=False, method="min").astype(int) 

16 df["rank_priority"] = df.priority_score.rank(ascending=False, method="min").astype(int) 

17 df["rank_gap"] = df.rank_pagerank - df.rank_priority 

18 

19 # Nodes a pure centrality ranking would put near the top that the rules demote. 

20 demoted = df[df.rank_pagerank <= top_n].nsmallest(top_n, "rank_gap") 

21 cols = ["gid", "role", "rank_pagerank", "rank_priority", "rank_gap", 

22 "in_deg", "out_deg", "in_kzt", "out_kzt", "pass_through", "evidence"] 

23 return demoted[cols].sort_values("rank_pagerank").reset_index(drop=True)
