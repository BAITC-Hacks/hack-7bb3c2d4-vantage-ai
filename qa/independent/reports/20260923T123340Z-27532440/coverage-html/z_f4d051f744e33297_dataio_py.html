1"""Loading and integrity checks. 

2 

3Every declared limitation from the case specification is detected here rather than 

4assumed, so the numbers in the README come from the data and not from the brief. 

5""" 

6from __future__ import annotations 

7 

8from dataclasses import dataclass 

9from pathlib import Path 

10 

11import networkx as nx 

12import pandas as pd 

13 

14 

15@dataclass 

16class Dataset: 

17 edges: pd.DataFrame 

18 nodes: pd.DataFrame 

19 tx: pd.DataFrame 

20 graph: nx.DiGraph 

21 

22 @property 

23 def seeds(self) -> set[int]: 

24 return set(self.nodes.loc[self.nodes.is_seed, "gid"]) 

25 

26 

27def load(data_dir: Path) -> Dataset: 

28 edges = pd.read_parquet(data_dir / "edges.parquet") 

29 nodes = pd.read_parquet(data_dir / "nodes.parquet") 

30 tx = pd.read_parquet(data_dir / "transactions.parquet") 

31 tx["date"] = pd.to_datetime(tx["date"]) 

32 

33 g = nx.DiGraph() 

34 # Nodes come from nodes.parquet, NOT from the edge list. 19 of the 81 seeds 

35 # appear in no edge at all; building from edges silently drops them and the 

36 # required output would have 2229 rows instead of 2248. 

37 for r in nodes.itertuples(index=False): 

38 g.add_node(int(r.gid), depth=int(r.depth), is_seed=bool(r.is_seed)) 

39 for r in edges.itertuples(index=False): 

40 g.add_edge(int(r.src), int(r.dst), 

41 sum_kzt=float(r.sum_kzt), n_tx=int(r.n_tx), depth=int(r.depth)) 

42 return Dataset(edges=edges, nodes=nodes, tx=tx, graph=g) 

43 

44 

45def integrity_report(d: Dataset) -> dict: 

46 """Measured facts about the declared limitations. Printed at run time and quoted in the README.""" 

47 g, nodes, edges, tx = d.graph, d.nodes, d.edges, d.tx 

48 in_edges = set(edges.src) | set(edges.dst) 

49 seeds = d.seeds 

50 out_deg = dict(g.out_degree()) 

51 depth = dict(zip(nodes.gid, nodes.depth)) 

52 

53 dead_ends = [n for n in g.nodes if out_deg.get(n, 0) == 0] 

54 truncated = [n for n in dead_ends if depth.get(n) == 4] 

55 genuine = [n for n in dead_ends if depth.get(n, 9) < 4] 

56 

57 agg = tx.groupby(["src", "dst"]).sum_kzt.agg(["sum", "size"]).reset_index() 

58 merged = edges.merge(agg, on=["src", "dst"], how="outer", indicator=True) 

59 

60 return { 

61 "nodes_declared": len(nodes), 

62 "nodes_in_graph": g.number_of_nodes(), 

63 "edges": g.number_of_edges(), 

64 "transactions": len(tx), 

65 "turnover_kzt": float(edges.sum_kzt.sum()), 

66 "period": (str(tx.date.min().date()), str(tx.date.max().date())), 

67 "seeds": len(seeds), 

68 "seeds_absent_from_edges": len(seeds - in_edges), 

69 "seeds_without_outgoing": sum(1 for s in seeds if out_deg.get(s, 0) == 0), 

70 "dead_ends": len(dead_ends), 

71 "dead_ends_truncated_depth4": len(truncated), 

72 "dead_ends_genuine": len(genuine), 

73 "weakly_connected_components": nx.number_weakly_connected_components(g), 

74 "edges_match_transactions": bool((merged._merge == "both").all()), 

75 }
