1"""What the crawl could not see, computed rather than disclaimed. 

2 

3The dataset is the result of a breadth-first crawl that followed OUTBOUND transfers four 

4hops from 81 seed accounts. Two consequences follow from that design alone, before any 

5metric is computed: 

6 

71. Money arriving INTO the seeds was never followed. An inbound transfer to a seed only 

8 produces a row when the payer itself was expanded by the crawl, and no account above 

9 the seeds ever was. Whoever funds the structure is therefore outside the dataset, and 

10 any confident org chart with a single figure at the top is describing the shape of the 

11 crawl rather than the shape of the group. 

122. A node at hop 4 looks like a terminal but is only the point at which the crawl stopped. 

13 Treating it as the same object as a node that genuinely received money and sent none is 

14 a factual error, and the two groups are different sizes. 

15 

16This module turns both into per-node fields and aggregate figures so that the uncertainty 

17travels with the evidence instead of sitting in a footnote. Nothing here writes a file. 

18"""

28# The crawl followed four hops. A node at this depth was recorded but never expanded. 

29CRAWL_DEPTH_LIMIT = 4 

30 

31# The five states a node can be in with respect to what the crawl actually observed. 

32# Evaluated in this order; the first rule that matches wins. 

33KNOWLEDGE_STATES = [ 

34 "isolated", # no transfer observed in either direction 

35 "truncated_at_depth", # sits at the crawl boundary, onward transfers never collected 

36 "outbound_only", # seed, or any node whose payer side was never followed 

37 "inbound_only", # money observed arriving, none observed leaving 

38 "fully_observed", # both directions traversed inside the crawl 

39] 

40 

41# Confidence is a weighted sum of five observation components, each bounded in [0, 1]. 

42# The weights sum to 1.0, so the result is bounded in [0, 1] without any clipping. 

43# They say how much of the reason to trust a node's role and rank is present in the data. 

44CONFIDENCE_WEIGHTS = { 

45 "outbound_enumerated": 0.34, # did the crawl expand this node and list where money went 

46 "inbound_observed": 0.26, # is any payer recorded, and is the payer side followable 

47 "downstream_resolved": 0.20, # how much of what is reachable below ends at the boundary 

48 "evidence_volume": 0.12, # enough transfers for a ratio to mean anything 

49 "depth_margin": 0.08, # distance from the boundary, as a proxy for how much 

50 # of the neighbourhood the crawl could still have reached 

51} 

52 

53# A ceiling per state, applied after the weighted sum. Without it an unexpanded hop-4 node 

54# with a rich inbound side scores respectably, which would misrepresent the one thing that 

55# actually matters about it: nobody looked at where its money went. 

56CONFIDENCE_CAPS = { 

57 "fully_observed": 1.00, 

58 "inbound_only": 0.85, 

59 "outbound_only": 0.60, 

60 "truncated_at_depth": 0.40, 

61 "isolated": 0.10, 

62} 

63 

64# A seed's recorded inbound edges are incidental return traffic from inside the crawl, not 

65# its funding side, so they earn partial rather than full credit on the inbound component. 

66SEED_INBOUND_CREDIT = 0.5 

67 

68# Below this, a node's role and rank should not be read without reading its limitation too. 

69CONFIDENCE_AUDIT_FLOOR = 0.75 

70 

71# The priority list an analyst would actually work through on day one. 

72TOP_LIST_SIZE = 20 

73 

74# Ranking rule for the next data request: the share of nodes a field would resolve plus the 

75# share of turnover it would illuminate. Stated here so the ordering is derived from the 

76# figures and can be recomputed by hand rather than argued about. 

77REQUEST_RANK_RULE = "share_of_nodes_resolved plus share_of_turnover_illuminated" 

78 

79# From docs/AGENT-BRIEF.md, verified against the raw data. If a figure below disagrees the 

80# summary says so in plain words rather than quietly adopting the computed value. 

81GROUND_TRUTH = { 

82 "seeds": 81, 

83 "seeds_isolated": 19, 

84 "seeds_without_outgoing": 31, 

85 "dead_ends": 1554, 

86 "dead_ends_truncated": 444, 

87 "dead_ends_genuine": 1110,

94def build(d: Dataset, feats: pd.DataFrame, roles_df: pd.DataFrame) -> tuple[pd.DataFrame, pd.DataFrame, dict]: 

95 """Per-node observation state, the ranked next data request, and the aggregate figures. 

96 

97 Returns (completeness_df, next_data_request_df, summary_dict). 

98 """ 

99 df = _assemble(feats, roles_df) 

100 df = _reachability(d, df) 

101 

102 df["hop_from_seed"] = df.depth.astype(int) 

103 df["inbound_known"] = df.in_deg > 0 

104 df["outbound_known"] = df.hop_from_seed < CRAWL_DEPTH_LIMIT 

105 df["knowledge_state"] = [_state(r) for r in df.itertuples(index=False)] 

106 df["confidence"] = [_confidence(r) for r in df.itertuples(index=False)] 

107 df["limitation"] = [_limitation(r) for r in df.itertuples(index=False)] 

108 

109 completeness = df[["gid", "knowledge_state", "hop_from_seed", "inbound_known", 

110 "outbound_known", "confidence", "limitation"]].copy() 

111 completeness = completeness.sort_values("gid").reset_index(drop=True) 

112 

113 summary = _summary(d, df) 

114 requests = _next_data_request(d, df, summary) 

115 return completeness, requests, summary

118def _assemble(feats: pd.DataFrame, roles_df: pd.DataFrame) -> pd.DataFrame: 

119 """One frame per node with whatever the caller has already computed, plus a priority score. 

120 

121 The caller may pass the roles frame before or after ranking. Where the rank is missing it 

122 is recomputed here with the same function the pipeline uses, so the audit of the top of 

123 the list is the audit of the real list and not of a second, different one. 

124 """ 

125 df = feats.copy() 

126 if roles_df is not None and id(roles_df) != id(feats): 

127 extra = [c for c in roles_df.columns if c == "gid" or c not in df.columns] 

128 if "gid" in roles_df.columns and len(extra) > 1: 

129 df = df.merge(roles_df[extra], on="gid", how="left") 

130 if "priority_score" not in df.columns: 

131 if "role" not in df.columns: 

132 raise ValueError("completeness.build needs a roles frame carrying a role column") 

133 df, _ = priority.rank(df) 

134 for col, default in (("in_tx", 0), ("out_tx", 0), ("in_kzt", 0.0), ("out_kzt", 0.0)): 

135 if col not in df.columns: 

136 df[col] = default 

137 return df

140def _reachability(d: Dataset, df: pd.DataFrame) -> pd.DataFrame: 

141 """For each node, how much of what it can reach sits at the crawl boundary. 

142 

143 A node whose downstream ends mostly at hop 4 has a role that is only as good as the hop 

144 the crawl stopped on, however clean its own arithmetic looks. Reachability is carried in 

145 integer bitmasks and iterated to a fixed point because the graph contains cycles, so a 

146 single topological pass would be wrong. 

147 """ 

148 gids = df.gid.tolist() 

149 index = {g: i for i, g in enumerate(gids)} 

150 n = len(gids) 

151 

152 successors: list[list[int]] = [[] for _ in range(n)] 

153 for r in d.edges.itertuples(index=False): 

154 src, dst = int(r.src), int(r.dst) 

155 if src in index and dst in index: 

156 successors[index[src]].append(index[dst]) 

157 

158 bit = [1 << i for i in range(n)] 

159 boundary_mask = 0 

160 for g, hop in zip(df.gid, df.depth): 

161 if int(hop) == CRAWL_DEPTH_LIMIT: 

162 boundary_mask |= bit[index[g]] 

163 

164 reach = [0] * n 

165 # Deepest first converges fastest: most of this graph flows from hop k to hop k+1. 

166 order = sorted(range(n), key=lambda i: -int(df.depth.iloc[i])) 

167 while True: 

168 changed = False 

169 for v in order: 

170 merged = reach[v] 

171 for u in successors[v]: 

172 merged |= bit[u] | reach[u] 

173 if merged != reach[v]: 

174 reach[v] = merged 

175 changed = True 

176 if not changed: 

177 break 

178 

179 desc, desc_boundary = [], [] 

180 for i in range(n): 

181 r = reach[i] & ~bit[i] # a node on a cycle reaches itself; that is not a descendant 

182 desc.append(r.bit_count()) 

183 desc_boundary.append((r & boundary_mask).bit_count()) 

184 

185 df = df.copy() 

186 df["descendants"] = desc 

187 df["descendants_truncated"] = desc_boundary 

188 share = np.where(np.array(desc) > 0, 

189 np.array(desc_boundary) / np.maximum(np.array(desc), 1), 0.0) 

190 # A node at the boundary has no observed downstream at all, which is unresolved rather 

191 # than resolved. Anything else with no successors was looked at and found empty. 

192 at_boundary = (df.depth == CRAWL_DEPTH_LIMIT).to_numpy() 

193 df["downstream_unresolved"] = np.where(at_boundary, 1.0, share) 

194 return df

197def _state(r) -> str: 

198 if r.in_deg == 0 and r.out_deg == 0: 

199 return "isolated" 

200 if r.hop_from_seed >= CRAWL_DEPTH_LIMIT: 

201 return "truncated_at_depth" 

202 # A seed is never fully observed however busy it looks. The crawl started here and only 

203 # ever followed money outwards, so its payer side is unobservable by construction. 

204 if r.out_deg > 0 and (bool(r.is_seed) or r.in_deg == 0): 

205 return "outbound_only" 

206 if r.out_deg == 0: 

207 return "inbound_only" 

208 return "fully_observed"

211def _confidence(r) -> float: 

212 w = CONFIDENCE_WEIGHTS 

213 

214 outbound = 1.0 if r.outbound_known else 0.0 

215 

216 if r.in_deg == 0: 

217 inbound = 0.0 

218 elif bool(r.is_seed): 

219 inbound = SEED_INBOUND_CREDIT 

220 else: 

221 inbound = 1.0 

222 

223 downstream = 1.0 - float(r.downstream_unresolved) 

224 

225 tx = int(r.in_tx) + int(r.out_tx) 

226 volume = min(1.0, tx / float(THRESHOLDS["min_tx_for_stable_ratio"])) 

227 

228 margin = (CRAWL_DEPTH_LIMIT - min(int(r.hop_from_seed), CRAWL_DEPTH_LIMIT)) / CRAWL_DEPTH_LIMIT 

229 

230 score = (w["outbound_enumerated"] * outbound 

231 + w["inbound_observed"] * inbound 

232 + w["downstream_resolved"] * downstream 

233 + w["evidence_volume"] * volume 

234 + w["depth_margin"] * margin) 

235 return round(min(score, CONFIDENCE_CAPS[r.knowledge_state]), 3)

242def _limitation(r) -> str: 

243 state = r.knowledge_state 

244 unresolved = int(round(float(r.downstream_unresolved) * 100)) 

245 

246 if state == "isolated": 

247 s = ("No transfer observed in either direction in the window, so nothing about this " 

248 "account's behaviour is measurable from this dataset.") 

249 elif state == "truncated_at_depth": 

250 s = (f"At the four-hop crawl boundary: {r.in_kzt:,.0f} KZT arrived from " 

251 f"{_payers(int(r.in_deg))} and where it went next was never collected.") 

252 elif state == "outbound_only": 

253 seen = (f"{_payers(int(r.in_deg))} appear only as return traffic from inside the crawl" 

254 if r.in_deg > 0 else "no payer appears at all") 

255 s = (f"Seed account: money out was followed, money in was not, so its funding is outside " 

256 f"the dataset ({seen}).") 

257 elif state == "inbound_only": 

258 if bool(r.is_seed): 

259 s = (f"Seed account with no observed outgoing transfer; the " 

260 f"{_payers(int(r.in_deg))} visible are return traffic from inside the crawl, " 

261 f"not its funding side.") 

262 else: 

263 s = (f"{r.in_kzt:,.0f} KZT arrived and none was observed leaving; any payer outside " 

264 f"the crawled set would not appear.") 

265 elif unresolved > 0: 

266 s = (f"Both directions observed, but {unresolved}% of what is reachable downstream ends " 

267 f"at the crawl boundary, so the onward path is unknown.") 

268 else: 

269 s = ("Both directions observed within the crawl; a payer outside the crawled set would " 

270 "still not appear in this dataset.") 

271 return s[:200]

274def _summary(d: Dataset, df: pd.DataFrame) -> dict: 

275 total_nodes = int(len(df)) 

276 turnover = float(d.edges.sum_kzt.sum()) 

277 

278 # Turnover touched by a node is what arrived plus what left. Summed over all nodes this is 

279 # twice the edge total, because each transfer is touched by its payer and by its receiver. 

280 df = df.copy() 

281 df["touched_kzt"] = df.in_kzt + df.out_kzt 

282 touched_total = float(df.touched_kzt.sum()) 

283 

284 out: dict = { 

285 "nodes_total": total_nodes, 

286 "edges_total": int(len(d.edges)), 

287 "transactions_total": int(len(d.tx)), 

288 "turnover_kzt": round(turnover, 2), 

289 "crawl_depth_limit": CRAWL_DEPTH_LIMIT, 

290 "crawl_direction": "outbound only from the seeds", 

291 } 

292 

293 for state in KNOWLEDGE_STATES: 

294 m = df.knowledge_state == state 

295 out[f"state_count_{state}"] = int(m.sum()) 

296 out[f"state_node_share_{state}"] = round(float(m.sum()) / total_nodes, 4) 

297 out[f"state_turnover_share_{state}"] = round( 

298 float(df.loc[m, "touched_kzt"].sum()) / touched_total, 4) if touched_total else 0.0 

299 

300 incomplete = df.knowledge_state != "fully_observed" 

301 out["nodes_incompletely_observed"] = int(incomplete.sum()) 

302 out["nodes_incompletely_observed_share"] = round(float(incomplete.sum()) / total_nodes, 4) 

303 out["turnover_share_incompletely_observed"] = round( 

304 float(df.loc[incomplete, "touched_kzt"].sum()) / touched_total, 4) if touched_total else 0.0 

305 

306 out["confidence_mean"] = round(float(df.confidence.mean()), 3) 

307 out["confidence_median"] = round(float(df.confidence.median()), 3) 

308 out["confidence_audit_floor"] = CONFIDENCE_AUDIT_FLOOR 

309 out["nodes_below_confidence_floor"] = int((df.confidence < CONFIDENCE_AUDIT_FLOOR).sum()) 

310 out["nodes_below_confidence_floor_share"] = round( 

311 float((df.confidence < CONFIDENCE_AUDIT_FLOOR).sum()) / total_nodes, 4) 

312 

313 # The number the demo turns on: how much of the list an analyst would work through first 

314 # is standing on something the crawl did not observe. 

315 # gid breaks ties so the audited list is the same list on every run. 

316 top = df.sort_values(["priority_score", "gid"], ascending=[False, True]).head(TOP_LIST_SIZE) 

317 resting = (top.knowledge_state != "fully_observed") | (top.confidence < CONFIDENCE_AUDIT_FLOOR) 

318 out["top_list_size"] = int(len(top)) 

319 out["top_nodes_resting_on_incomplete_observation"] = int(resting.sum()) 

320 out["top_nodes_resting_share"] = round(float(resting.sum()) / max(len(top), 1), 4) 

321 out["top_nodes_that_are_seeds"] = int(top.is_seed.sum()) 

322 out["top_nodes_with_truncated_descendants"] = int((top.descendants_truncated > 0).sum()) 

323 out["top_nodes_mean_downstream_unresolved_share"] = round( 

324 float(top.downstream_unresolved.mean()), 4) 

325 out["top_nodes_confidence_mean"] = round(float(top.confidence.mean()), 3) 

326 out["top_nodes_confidence_min"] = round(float(top.confidence.min()), 3) 

327 

328 seeds = df[df.is_seed] 

329 seed_out = float(seeds.out_kzt.sum()) 

330 seed_in = float(seeds.in_kzt.sum()) 

331 unattributed = seed_out - seed_in 

332 out["seeds_total"] = int(len(seeds)) 

333 out["seeds_isolated"] = int(seeds.isolated.sum()) if "isolated" in seeds.columns else int( 

334 ((seeds.in_deg == 0) & (seeds.out_deg == 0)).sum()) 

335 out["seeds_without_outgoing"] = int((seeds.out_deg == 0).sum()) 

336 out["seeds_with_incidental_inbound"] = int((seeds.in_deg > 0).sum()) 

337 out["seed_outbound_kzt"] = round(seed_out, 2) 

338 out["seed_inbound_observed_kzt"] = round(seed_in, 2) 

339 out["seed_funding_unattributed_kzt"] = round(unattributed, 2) 

340 out["seed_funding_unattributed_share_of_turnover"] = round(unattributed / turnover, 4) if turnover else 0.0 

341 out["seed_inbound_crawled"] = 0 

342 out["seed_funding_visibility"] = ( 

343 "not observable: a transfer into a seed is recorded only if the payer was expanded by " 

344 "the crawl, and no account above the seeds ever was") 

345 # Interpolated, not written in: every figure in a generated string comes from the value 

346 # computed beside it, so a different extract cannot leave a stale number in the prose. 

347 out["blind_spot_statement"] = ( 

348 f"Inbound transfers to the {out['seeds_total']} seed accounts were never followed, so the " 

349 f"funding source of the structure lies outside this dataset") 

350 

351 boundary = df[df.knowledge_state == "truncated_at_depth"] 

352 out["truncated_nodes"] = int(len(boundary)) 

353 out["truncated_inbound_kzt"] = round(float(boundary.in_kzt.sum()), 2) 

354 out["truncated_inbound_share_of_turnover"] = round( 

355 float(boundary.in_kzt.sum()) / turnover, 4) if turnover else 0.0 

356 

357 dead_ends = df[df.out_deg == 0] 

358 out["dead_ends_total"] = int(len(dead_ends)) 

359 out["dead_ends_truncated"] = int((dead_ends.depth == CRAWL_DEPTH_LIMIT).sum()) 

360 out["dead_ends_genuine"] = int((dead_ends.depth < CRAWL_DEPTH_LIMIT).sum()) 

361 

362 out["ground_truth_check"] = _ground_truth_check(out) 

363 return out

366def _ground_truth_check(out: dict) -> str: 

367 """Compare against the brief. A disagreement is reported, never absorbed.""" 

368 computed = { 

369 "seeds": out["seeds_total"], 

370 "seeds_isolated": out["seeds_isolated"], 

371 "seeds_without_outgoing": out["seeds_without_outgoing"], 

372 "dead_ends": out["dead_ends_total"], 

373 "dead_ends_truncated": out["dead_ends_truncated"], 

374 "dead_ends_genuine": out["dead_ends_genuine"], 

375 } 

376 bad = [f"{k}: computed {computed[k]}, brief says {v}" 

377 for k, v in GROUND_TRUTH.items() if computed[k] != v] 

378 return "agrees with the brief" if not bad else "DISAGREES WITH THE BRIEF: " + "; ".join(bad)

381def _next_data_request(d: Dataset, df: pd.DataFrame, summary: dict) -> pd.DataFrame: 

382 """What to ask for next, ordered by what the present figures say each would buy. 

383 

384 Every row carries a count of nodes it would move out of an uncertain state and a KZT figure 

385 it would put an interpretation on. A request the three tables give no way to evaluate is 

386 kept in the list and marked as such rather than dropped, because a silent omission would 

387 read as a judgement the data does not support. 

388 """ 

389 turnover = float(d.edges.sum_kzt.sum()) 

390 total_nodes = int(len(df)) 

391 tx = d.tx 

392 

393 # Day-level dates leave the order of arrival and departure unresolved whenever both happen 

394 # on the same calendar day, which is exactly where the transit reading is decided. 

395 arrivals = tx.groupby(["dst", "date"], as_index=False).sum_kzt.sum().rename(columns={"dst": "gid"}) 

396 departures = tx.groupby(["src", "date"], as_index=False).size().rename(columns={"src": "gid"}) 

397 ambiguous = arrivals.merge(departures[["gid", "date"]], on=["gid", "date"], how="inner") 

398 ambiguous_nodes = int(ambiguous.gid.nunique()) 

399 ambiguous_kzt = float(ambiguous.sum_kzt.sum()) 

400 

401 # An account whose entire observed life is one calendar day, with money both in and out, is 

402 # the shape an opening date would settle: purpose-opened, or long-standing and used once. 

403 first = pd.concat([tx.groupby("src").date.min().rename("a"), 

404 tx.groupby("dst").date.min().rename("b")], axis=1).min(axis=1) 

405 last = pd.concat([tx.groupby("src").date.max().rename("a"), 

406 tx.groupby("dst").date.max().rename("b")], axis=1).max(axis=1) 

407 span_days = (last - first).dt.days 

408 single_day = set(span_days[span_days == 0].index) 

409 short_life = df[df.gid.isin(single_day) & (df.in_deg > 0) & (df.out_deg > 0)] 

410 short_life_nodes = int(len(short_life)) 

411 short_life_kzt = float(short_life.in_kzt.sum()) 

412 

413 rows = [ 

414 { 

415 "request": (f"One more crawl hop from the {summary['truncated_nodes']:,} nodes at " 

416 f"the depth limit"), 

417 "why": (f"{summary['truncated_nodes']} nodes are dead ends only because the crawl " 

418 f"stopped at hop {CRAWL_DEPTH_LIMIT}. {summary['truncated_inbound_kzt']:,.0f} KZT " 

419 f"arrives at them and its onward path is unrecorded, so they cannot be told " 

420 f"apart from the {summary['dead_ends_genuine']:,} nodes where money " 

421 f"genuinely appears to stop."), 

422 "resolves_nodes": summary["truncated_nodes"], 

423 "illuminates_kzt": summary["truncated_inbound_kzt"], 

424 "feasibility_note": ("Same collection method, one hop further. Node count grows with " 

425 "the fan-out already seen at hop 3, so scope the extraction first."), 

426 }, 

427 { 

428 "request": "Hour-level timestamps in place of calendar dates", 

429 "why": (f"{ambiguous_nodes} nodes have at least one day on which money both arrived and " 

430 f"left, covering {ambiguous_kzt:,.0f} KZT. At day resolution the order of those " 

431 f"movements is unresolved, and dwell time is what separates a node passing funds " 

432 f"through from one that happened to transact twice."), 

433 "resolves_nodes": ambiguous_nodes, 

434 "illuminates_kzt": round(ambiguous_kzt, 2), 

435 "feasibility_note": ("The field almost certainly exists in the source ledger and was " 

436 "reduced on export; a re-export would not need a new crawl."), 

437 }, 

438 { 

439 "request": (f"Inbound transfers to the {summary['seeds_total']} seed accounts, " 

440 f"one hop back"), 

441 "why": (f"Seeds moved {summary['seed_outbound_kzt']:,.0f} KZT out against " 

442 f"{summary['seed_inbound_observed_kzt']:,.0f} KZT observed arriving, leaving " 

443 f"{summary['seed_funding_unattributed_kzt']:,.0f} KZT with no recorded payer. " 

444 f"This is the only request that can reach above the seeds at all."), 

445 "resolves_nodes": int(summary["seeds_total"]), 

446 "illuminates_kzt": summary["seed_funding_unattributed_kzt"], 

447 "feasibility_note": ("Reverses the crawl direction, so it is a new collection rather " 

448 "than a re-export. The unattributed figure assumes seeds opened " 

449 "the window at zero; the data carries no balances, so read it as " 

450 "the amount without an observed payer, not as proven outside money."), 

451 }, 

452 { 

453 "request": "Account opening dates", 

454 "why": (f"{short_life_nodes} accounts show their whole observed life inside one calendar " 

455 f"day with money both in and out, covering {short_life_kzt:,.0f} KZT. An opening " 

456 f"date would separate an account opened for the purpose from a long-standing one " 

457 f"used once, which the transfer record on its own cannot."),
