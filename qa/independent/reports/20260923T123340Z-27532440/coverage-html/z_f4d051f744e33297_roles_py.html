1"""The rule bank. 

2 

3Every role is a named rule with numeric thresholds. Nothing here is a model call: a 

4role can be traced to a line in this file and re-checked by hand, which is what the 

5case specification requires under must have 3. 

6 

7Thresholds live in THRESHOLDS so the README table and the code cannot drift apart. 

8 

9Two of the eight roles are abstentions rather than behavioural claims, and they are named 

10so that a reader cannot mistake either one for a gap. 

11 

12`abstained_boundary` is the outcome for the 444 nodes that sit at hop 4 with no outgoing 

13edge. Their pass-through ratio is zero because the crawl stopped there, not because they 

14retained the money, so applying the consolidator rule to them would read a collection 

15artefact as behaviour. None of them reaches the consolidator fan-in floor of 8 in any case: 

16434 have an in-degree of exactly 1, seven have 2 and three have 3. There is no defensible 

17rule left to write about them, so the system declines to write one. 

18 

19`abstained_single_observation` is the outcome for the 114 nodes with exactly one transfer in 

20and one out. A pass-through figure built from a single pair of numbers is not a rate, and 

21cannot be told apart from coincidence. 

22 

23That abstention sits below the transit rule in the precedence order rather than above it, 

24so 27 of the 72 transit nodes are themselves built on one transfer each way. Those 27 say 

25so in their own evidence. The reasoning for keeping the order this way is written above the 

26transit branch. 

27 

28Both are deliberate and both can say why for every row they cover, which the evidence column 

29carries with the figures the abstention rests on. The two names are also the keys shared with 

30the review screen's palette, docs/COMPLETENESS.md and the role figure, so they are changed in 

31all four places together or not at all. 

32""" 

33from __future__ import annotations 

34 

35import pandas as pd 

36 

37ROLES = ["consolidator", "transit", "distributor", "terminal", 

38 "coordinator", "peripheral", 

39 "abstained_boundary", "abstained_single_observation"] 

40 

41# Each cutoff is placed against the observed distribution rather than picked for being a 

42# round number. Shares below are measured over the 2,229 nodes that appear in at least one 

43# edge; the remaining 19 are seeds with no transfers in the window. 

44THRESHOLDS = { 

45 # The case notes declare a fan-in band of 8 to 24 distinct payers, and the data agrees: 

46 # the observed maximum is exactly 24, and 17 nodes (0.76%, the 99.5th percentile) reach 

47 # 8 against 5 more sitting at 7. The cutoff is the case's own floor, confirmed here. 

48 "consolidator_min_in_deg": 8, 

49 # Retention rather than relay. 0.5 sits well clear of the 0.8 floor of the transit band, 

50 # so no node can satisfy both rules and the precedence order cannot hide a near miss. 

51 "consolidator_max_pass_through": 0.5, 

52 # 20 falls in an empty bin: no node has out_deg of exactly 20, the observed values run 

53 # 19, 19, 19, then 21, 21, 21. The cut is therefore a real gap rather than a round 

54 # number. 28 nodes (1.26%) sit above it, and the 60 to 116 fan-out band the case notes 

55 # describe is the top 8 of those. 

56 "distributor_min_out_deg": 20, 

57 # The band that reproduces the 72 pass-through nodes the case declares, exactly. 

58 # Tightening it to 0.9 to 1.1 leaves 51 and would contradict that declared figure. 

59 "transit_pass_through_lo": 0.8, 

60 "transit_pass_through_hi": 1.2, 

61 # The weakest cutoff here, and stated as such: 4 sits on a smooth decay rather than on a 

62 # gap, with 90 nodes active on both sides at 3, 36 at 4 and 18 at 5. It is the 97.5th 

63 # percentile of in-degree and the 93rd of out-degree, so it selects nodes well above the 

64 # typical one without claiming the data shows a natural break. 

65 "coordinator_min_in_deg": 4, 

66 "coordinator_min_out_deg": 4, 

67 # One transfer in and one out is a single observation, not a rate: a pass-through figure 

68 # built from it cannot be told apart from coincidence. 114 nodes reach this rule. 

69 "min_tx_for_stable_ratio": 3, 

70}

73def assign(df: pd.DataFrame) -> pd.DataFrame: 

74 """Returns df with role, role_score and evidence. Order of rules is the precedence order.""" 

75 out = df.copy() 

76 out["role"] = "" 

77 out["role_score"] = 0.0 

78 out["evidence"] = "" 

79 

80 for i, r in out.iterrows(): 

81 role, score, ev = _classify(r) 

82 out.at[i, "role"] = role 

83 out.at[i, "role_score"] = round(score, 3) 

84 out.at[i, "evidence"] = ev[:200] 

85 return out

88def _payers(n: int) -> str: 

89 return f"{n} payer" if n == 1 else f"{n} payers"

92def _classify(r) -> tuple[str, float, str]: 

93 T = THRESHOLDS 

94 

95 # Scored at the bottom of the bank rather than the top. Nothing was observed moving in 

96 # either direction, so the peripheral label rests on an absence and not on a measurement. 

97 # 0.0 is reserved for the two abstention classes, where a rule was declined outright; 

98 # here a rule did fire, but on no transfer evidence at all, so it sits just above them 

99 # and below the 0.4 a peripheral account with observed transfers earns. 

100 if r.isolated: 

101 known = "already known to the investigation" if bool(r.is_seed) else "not a known client" 

102 return ("peripheral", 0.1, 

103 f"no transfer observed in either direction in the window, so nothing about this " 

104 f"account's behaviour is measurable here; depth={int(r.depth)}, {known}") 

105 

106 # Declared limitation: seed inflow is understated by the crawl design, so any rule 

107 # that reads in_kzt for a seed is unsound. Seeds are judged on outbound only. 

108 if r.is_seed and r.in_deg == 0 and r.out_deg > 0: 

109 return ("peripheral", 0.5, 

110 f"seed, outbound only: out_deg={int(r.out_deg)}, out={r.out_kzt:,.0f} KZT; " 

111 f"inflow not observable by collection design") 

112 

113 if r.truncated_by_depth: 

114 return ("abstained_boundary", 0.0, 

115 f"abstained at the crawl boundary: hop 4 with no outgoing edge, so nothing " 

116 f"downstream was collected; in={r.in_kzt:,.0f} KZT from {_payers(int(r.in_deg))}") 

117 

118 if r.genuine_terminal: 

119 return ("terminal", 0.8, 

120 f"depth={int(r.depth)} < 4 and no outgoing: money arrives and stays; " 

121 f"in={r.in_kzt:,.0f} KZT over {int(r.in_tx)} transfers from {_payers(int(r.in_deg))}") 

122 

123 # The four structural rules. Precedence runs widest fan-in first, then widest fan-out, 

124 # then relay ratio, then two-sided activity, because a node can satisfy more than one and 

125 # the earlier rule is the stronger claim about it. Each cutoff is justified in THRESHOLDS. 

126 if r.in_deg >= T["consolidator_min_in_deg"] and ( 

127 pd.isna(r.pass_through) or r.pass_through <= T["consolidator_max_pass_through"]): 

128 ratio = ("inflow only, no outgoing ratio" if pd.isna(r.pass_through) 

129 else f"pass_through={r.pass_through:.2f}") 

130 return ("consolidator", 0.7, 

131 f"in_deg={_payers(int(r.in_deg))}, in={r.in_kzt:,.0f} KZT, {ratio}") 

132 

133 if r.out_deg >= T["distributor_min_out_deg"]: 

134 return ("distributor", 0.7, 

135 f"out_deg={int(r.out_deg)} receivers, out={r.out_kzt:,.0f} KZT " 

136 f"over {int(r.out_tx)} transfers") 

137 

138 # Dwell time is reported beside the ratio but is not a gate. Only 60 of these 72 nodes 

139 # move money on within two days, so gating on it would contradict the 72 pass-through 

140 # nodes the case declares, and 28 of the 72 show a negative dwell because outflow was 

141 # observed before any inflow was, which is the outbound-only crawl rather than a real 

142 # ordering. The figure is left in the evidence for an analyst to weigh. 

143 # Precedence, decided deliberately: `min_tx_for_stable_ratio` sits below this branch 

144 # rather than above it. The case declares 72 pass-through accounts and this band 

145 # reproduces that figure exactly; gating transit on the transfer count first would drop 

146 # it to 45 and contradict a figure the case itself states. 27 of the 72 have one 

147 # transfer in and one out, so their ratio is a single observation rather than a rate. 

148 # Those 27 are neither silently promoted nor silently dropped: they keep the role and 

149 # the evidence below says on its face what the ratio rests on, so an analyst reading 

150 # the row can discount it without having to re-derive the transfer counts. 

151 if pd.notna(r.pass_through) and T["transit_pass_through_lo"] <= r.pass_through <= T["transit_pass_through_hi"]: 

152 dwell = "" if pd.isna(r.dwell_days) else f", dwell={int(r.dwell_days)}d" 

153 if (r.in_tx + r.out_tx) < T["min_tx_for_stable_ratio"]: 

154 return ("transit", 0.7, 

155 f"pass-through {r.pass_through:.2f} from a single transfer each way, so the " 

156 f"ratio is one observation rather than a rate: in={r.in_kzt:,.0f}, " 

157 f"out={r.out_kzt:,.0f} KZT{dwell}") 

158 return ("transit", 0.7, 

159 f"pass_through={r.pass_through:.2f}: in={r.in_kzt:,.0f}, out={r.out_kzt:,.0f} KZT{dwell}") 

160 

161 if r.in_deg >= T["coordinator_min_in_deg"] and r.out_deg >= T["coordinator_min_out_deg"]: 

162 return ("coordinator", 0.5, 

163 f"both sides active: in_deg={int(r.in_deg)}, out_deg={int(r.out_deg)}, " 

164 f"in={r.in_kzt:,.0f}, out={r.out_kzt:,.0f} KZT") 

165 

166 if (r.in_tx + r.out_tx) < T["min_tx_for_stable_ratio"]: 

167 return ("abstained_single_observation", 0.0, 

168 f"abstained on a single observation: in_tx={int(r.in_tx)}, out_tx={int(r.out_tx)}, " 

169 f"fewer than {T['min_tx_for_stable_ratio']} transfers is not a stable ratio") 

170 

171 return ("peripheral", 0.4, 

172 f"no rule threshold met: in_deg={int(r.in_deg)}, out_deg={int(r.out_deg)}, " 

173 f"in={r.in_kzt:,.0f}, out={r.out_kzt:,.0f} KZT")
